cmake_minimum_required(VERSION 3.20)
project(LodPatchTests LANGUAGES CXX)
enable_testing()
add_executable(lod_patch_test lod_patch_test.cpp)
target_include_directories(lod_patch_test PRIVATE ../../include)
target_compile_features(lod_patch_test PRIVATE cxx_std_20)
target_compile_definitions(lod_patch_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set(CVR_TEST_GAME_EXE "" CACHE FILEPATH "Cyberpunk2077.exe to validate without launching it")
if(CVR_TEST_GAME_EXE)
    add_test(NAME lod_native_tail_flags_and_guards COMMAND lod_patch_test "${CVR_TEST_GAME_EXE}")
    set_tests_properties(lod_native_tail_flags_and_guards PROPERTIES TIMEOUT 30)
endif()
