cmake_minimum_required(VERSION 3.24)
project(ExternalCameraTests LANGUAGES CXX)
enable_testing()
add_executable(non_fpp_tests main.cpp)
target_compile_features(non_fpp_tests PRIVATE cxx_std_20)
target_include_directories(non_fpp_tests PRIVATE ../../include ../../build/_deps/openxr-src/include)
if(MSVC)
    target_compile_options(non_fpp_tests PRIVATE /W4 /permissive-)
endif()
foreach(case neutral tracking stereo blend invalid fixed_point repeated source_unchanged lease panzer_forward aim_centre aim_scope aim_direction)
    add_test(NAME non_fpp_${case} COMMAND non_fpp_tests ${case})
endforeach()
