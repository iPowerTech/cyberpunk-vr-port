cmake_minimum_required(VERSION 3.24)
project(StartupHookTests LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/generate_fixture.py"
    "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
foreach(variant current regression)
    add_executable(hook_${variant} "${CMAKE_CURRENT_BINARY_DIR}/${variant}.cpp")
    target_compile_features(hook_${variant} PRIVATE cxx_std_20)
endforeach()
add_test(NAME startup_publish_before_patch COMMAND hook_current)
add_test(NAME startup_protection_failure COMMAND hook_current failed_protection)
add_test(NAME startup_old_publication_regression COMMAND hook_regression)
set_tests_properties(startup_old_publication_regression PROPERTIES WILL_FAIL TRUE)
