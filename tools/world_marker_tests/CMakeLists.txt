cmake_minimum_required(VERSION 3.24)
project(WorldMarkerTests LANGUAGES CXX ASM_MASM)
enable_testing()
add_executable(world_marker_tests main.cpp abi.asm
    ../../src/Render/WorldMarkerShader.cpp ../../src/Hooks/WorldMarkerTextThunk.asm)
include(../../cmake/WorldMarkerShaders.cmake)
cvr_world_marker_shaders(world_marker_tests)
target_include_directories(world_marker_tests PRIVATE ../../include)
target_compile_features(world_marker_tests PRIVATE cxx_std_20)
target_compile_definitions(world_marker_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(world_marker_tests PRIVATE d3d12 dxgi d3dcompiler)
foreach(case depth camera marker_size sprite procedural text text_batched abi effect_owner)
    add_test(NAME world_marker_${case} COMMAND world_marker_tests ${case})
endforeach()
set(native_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../build/world-markers-re/shaders")
if(EXISTS "${native_dir}/shader_104_20.dxbc" AND EXISTS "${native_dir}/shader_92_2.dxbc")
    add_test(NAME world_marker_native_parity COMMAND world_marker_tests native "${native_dir}")
endif()
if(EXISTS "${native_dir}/shader_104_20.dxbc" AND EXISTS "${native_dir}/shader_92_2.dxbc"
   AND EXISTS "${native_dir}/shader_104_21.dxbc" AND EXISTS "${native_dir}/shader_92_3.dxbc")
    add_test(NAME world_marker_native_link COMMAND world_marker_tests native_link "${native_dir}")
endif()
